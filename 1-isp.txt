hostnamectl set-hostname isp
timedatectl set-timezone Europe/Moscow

# Настройка NAT
# -s адреса ens4 ens5 т.е. роутеры
# -o ens3 это выход в интернет
iptables -t nat -A POSTROUTING -s 172.16.1.0/28 -o ens3 -j MASQUERADE
iptables -t nat -A POSTROUTING -s 172.16.2.0/28 -o ens3 -j MASQUERADE

# Сохранить iptables
iptables-save > /etc/sysconfig/iptables
# запустить iptables
systemctl start --now iptables.service

# доп проверка ISP
hostname
timedatectl | grep "Time zone"
ip -br a
iptables -t nat -L POSTROUTING -n -v
